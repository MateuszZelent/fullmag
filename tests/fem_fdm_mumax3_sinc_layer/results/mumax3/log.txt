//output directory: /tmp/fullmag-sinc-layer-mumax3-manual/
fcut := 10e9
t0 := 20 / fcut
SetGridSize(200, 200, 1)
SetCellSize(2.5e-9, 2.5e-9, 10e-9)
SetPBC(0, 0, 0)
//resizing...
Msat = 800e3
Aex = 13e-12
alpha = 0.01
Temp = 0
m = uniform(1, 0, 0)
B_ext = vector(100e-3, 1e-3*sinc(2*pi*fcut*(t-t0)), 0)
MaxErr = 1e-7
relax()
tableadd(m)
tableadd(E_exch)
tableadd(E_demag)
tableadd(E_zeeman)
tableadd(E_anis)
tableadd(E_total)
tableautosave(1 / (2 * 1.3 * fcut))
run(40 / fcut)
